### 读占空比和频率

```
void HAL_TIM_IC_CaptureCallback(TIM_HandleTypeDef *htim)
{
	if(htim->Instance==TIM2)
	{
		if(htim->Channel==HAL_TIM_ACTIVE_CHANNEL_2)//直接输入通道
		{
			ccrl_vala = HAL_TIM_ReadCapturedValue(htim,TIM_CHANNEL_2);
			ccrl_valb = HAL_TIM_ReadCapturedValue(htim,TIM_CHANNEL_1);
			__HAL_TIM_SetCounter(htim,0);
			frq=(800000000/80)/ccrl_vala;
			duty=(ccrl_valb/ccrl_vala)*100;
			
			HAL_TIM_IC_Start_IT(htim,TIM_CHANNEL_1);
			HAL_TIM_IC_Start_IT(htim,TIM_CHANNEL_2);
		}
	}
}
```









### 读ADC

```
double adc_get(ADC_HandleTypeDef *hadc)
{
	uint adc;
	HAL_ADC_Start(hadc);
	adc = HAL_ADC_GetValue(hadc);
	return adc*3.3/4096;
}
```
